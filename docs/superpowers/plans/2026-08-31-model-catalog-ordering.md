- [ ] **Step 2: 实现最小稳定排序函数**

```js
export function orderModelEntries(entries, { categoryOrder, staticUrlPrefix }) {
  const rankByCategory = new Map(categoryOrder.map((category, index) => [category, index]));
  return entries
    .map((entry, originalIndex) => ({ entry, originalIndex, isStatic: entry.fileUrl.startsWith(staticUrlPrefix) }))
    .sort((left, right) => {
      if (left.isStatic !== right.isStatic) return left.isStatic ? -1 : 1;
      if (!left.isStatic) return left.originalIndex - right.originalIndex;

      const leftRank = rankByCategory.get(left.entry.category);
      const rightRank = rankByCategory.get(right.entry.category);
      if (leftRank === undefined) {
        throw new Error(`unknown model category for ${left.entry.id}: ${left.entry.category}`);
      }
      if (rightRank === undefined) {
        throw new Error(`unknown model category for ${right.entry.id}: ${right.entry.category}`);
      }
      return leftRank - rightRank || left.originalIndex - right.originalIndex;
    })
    .map(({ entry }) => entry);
}
```
